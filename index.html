const express = require('express'); const cors = require('cors'); const fs = require('fs'); const path = require('path'); const app = express(); app.use(cors({ origin: '*' })); app.use(express.json()); const DB_FILE = path.join(__dirname, 'notes.json'); const ADMIN_PASSWORD = "mojeTajneHeslo123"; // Tvoje tajné heslo function getNotes() { if (!fs.existsSync(DB_FILE)) { fs.writeFileSync(DB_FILE, JSON.stringify([])); } const data = fs.readFileSync(DB_FILE, 'utf8'); return JSON.parse(data || '[]'); } function saveNotes(notes) { fs.writeFileSync(DB_FILE, JSON.stringify(notes, null, 2)); } // 1. Overenie admin hesla app.post('/api/verify-admin', (req, res) => { const { password } = req.body; if (password === ADMIN_PASSWORD) { res.json({ success: true }); } else { res.status(403).json({ error: 'Nesprávne heslo!' }); } }); // 2. Získanie poznámok app.get('/api/notes', (req, res) => { try { const notes = getNotes(); res.json(notes); } catch (err) { res.status(500).json({ error: err.message }); } }); // 3. Pridanie poznámky app.post('/api/notes', (req, res) => { const { title, content, class: className, subject, file_url, is_pinned, password } = req.body; if (password !== ADMIN_PASSWORD) { return res.status(403).json({ error: 'Nesprávne admin heslo!' }); } if (!title || !content) { return res.status(400).json({ error: 'Názov aj obsah sú povinné!' }); } try { const notes = getNotes(); const newNote = { id: Date.now(), title, content, class: className || '1.A', subject: subject || 'Všeobecné', file_url: file_url || '', is_pinned: !!is_pinned, created_at: new Date().toISOString() }; notes.unshift(newNote); saveNotes(notes); res.json(newNote); } catch (err) { res.status(500).json({ error: err.message }); } }); // 4. Úprava poznámky (PUT) app.put('/api/notes/:id', (req, res) => { const { title, content, class: className, subject, file_url, is_pinned, password } = req.body; const noteId = parseInt(req.params.id); if (password !== ADMIN_PASSWORD) { return res.status(403).json({ error: 'Nesprávne admin heslo!' }); } try { let notes = getNotes(); const index = notes.findIndex(n => n.id === noteId); if (index === -1) { return res.status(404).json({ error: 'Poznámka sa nenenašla.' }); } notes[index] = { ...notes[index], title: title || notes[index].title, content: content || notes[index].content, class: className || notes[index].class, subject: subject || notes[index].subject, file_url: file_url !== undefined ? file_url : notes[index].file_url, is_pinned: is_pinned !== undefined ? !!is_pinned : notes[index].is_pinned, updated_at: new Date().toISOString() }; saveNotes(notes); res.json(notes[index]); } catch (err) { res.status(500).json({ error: err.message }); } }); // 5. Zmazanie poznámky app.delete('/api/notes/:id', (req, res) => { const { password } = req.body; const noteId = parseInt(req.params.id); if (password !== ADMIN_PASSWORD) { return res.status(403).json({ error: 'Nesprávne admin heslo!' }); } try { let notes = getNotes(); notes = notes.filter(n => n.id !== noteId); saveNotes(notes); res.json({ message: 'Poznámka bola zmazaná.' }); } catch (err) { res.status(500).json({ error: err.message }); } }); const PORT = process.env.PORT || 10000; app.listen(PORT, () => console.log(`Server beží na porte ${PORT}`));
